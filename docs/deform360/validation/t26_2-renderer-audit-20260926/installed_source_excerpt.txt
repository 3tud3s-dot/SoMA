74: __device__ float3 computeCov2D(const float3& mean, float focal_x, float focal_y, float tan_fovx, float tan_fovy, const float* cov3D, const float* viewmatrix)
75: {
76: 	// The following models the steps outlined by equations 29
77: 	// and 31 in "EWA Splatting" (Zwicker et al., 2002). 
78: 	// Additionally considers aspect / scaling of viewport.
79: 	// Transposes used to account for row-/column-major conventions.
80: 	float3 t = transformPoint4x3(mean, viewmatrix);
81: 
82: 	const float limx = 1.3f * tan_fovx;
83: 	const float limy = 1.3f * tan_fovy;
84: 	const float txtz = t.x / t.z;
85: 	const float tytz = t.y / t.z;
86: 	t.x = min(limx, max(-limx, txtz)) * t.z;
87: 	t.y = min(limy, max(-limy, tytz)) * t.z;
88: 
89: 	glm::mat3 J = glm::mat3(
90: 		focal_x / t.z, 0.0f, -(focal_x * t.x) / (t.z * t.z),
91: 		0.0f, focal_y / t.z, -(focal_y * t.y) / (t.z * t.z),
92: 		0, 0, 0);
93: 
94: 	glm::mat3 W = glm::mat3(
95: 		viewmatrix[0], viewmatrix[4], viewmatrix[8],
96: 		viewmatrix[1], viewmatrix[5], viewmatrix[9],
97: 		viewmatrix[2], viewmatrix[6], viewmatrix[10]);
98: 
99: 	glm::mat3 T = W * J;
100: 
101: 	glm::mat3 Vrk = glm::mat3(
102: 		cov3D[0], cov3D[1], cov3D[2],
103: 		cov3D[1], cov3D[3], cov3D[4],
104: 		cov3D[2], cov3D[4], cov3D[5]);
105: 
106: 	glm::mat3 cov = glm::transpose(T) * glm::transpose(Vrk) * T;
107: 
108: 	return { float(cov[0][0]), float(cov[0][1]), float(cov[1][1]) };
109: }

111: // Forward method for converting scale and rotation properties of each
112: // Gaussian to a 3D covariance matrix in world space. Also takes care
113: // of quaternion normalization.
202: 	if (cov3D_precomp != nullptr)
203: 	{
204: 		cov3D = cov3D_precomp + idx * 6;
205: 	}
206: 	else
207: 	{
208: 		computeCov3D(scales[idx], scale_modifier, rotations[idx], cov3Ds + idx * 6);
209: 		cov3D = cov3Ds + idx * 6;
210: 	}
211: 
212: 	// Compute 2D screen-space covariance matrix
213: 	float3 cov = computeCov2D(p_orig, focal_x, focal_y, tan_fovx, tan_fovy, cov3D, viewmatrix);
214: 
215: 	constexpr float h_var = 0.3f;
216: 	const float det_cov = cov.x * cov.z - cov.y * cov.y;
217: 	cov.x += h_var;
218: 	cov.z += h_var;
219: 	const float det_cov_plus_h_cov = cov.x * cov.z - cov.y * cov.y;
220: 	float h_convolution_scaling = 1.0f;
221: 
222: 	if(antialiasing)
223: 		h_convolution_scaling = sqrt(max(0.000025f, det_cov / det_cov_plus_h_cov)); // max for numerical stability
224: 
225: 	// Invert covariance (EWA algorithm)
226: 	const float det = det_cov_plus_h_cov;
227: 
228: 	if (det == 0.0f)
229: 		return;
230: 	float det_inv = 1.f / det;
231: 	float3 conic = { cov.z * det_inv, -cov.y * det_inv, cov.x * det_inv };
232: 
233: 	// Compute extent in screen space (by finding eigenvalues of
234: 	// 2D covariance matrix). Use extent to compute a bounding rectangle
235: 	// of screen-space tiles that this Gaussian overlaps with. Quit if
236: 	// rectangle covers 0 tiles. 
237: 	float mid = 0.5f * (cov.x + cov.z);
238: 	float lambda1 = mid + sqrt(max(0.1f, mid * mid - det));
239: 	float lambda2 = mid - sqrt(max(0.1f, mid * mid - det));
240: 	float my_radius = ceil(3.f * sqrt(max(lambda1, lambda2)));
241: 	float2 point_image = { ndc2Pix(p_proj.x, W), ndc2Pix(p_proj.y, H) };
242: 	uint2 rect_min, rect_max;
243: 	getRect(point_image, my_radius, rect_min, rect_max, grid);
244: 	if ((rect_max.x - rect_min.x) * (rect_max.y - rect_min.y) == 0)
245: 		return;
246: 
247: 	// If colors have been precomputed, use them, otherwise convert
248: 	// spherical harmonics coefficients to RGB color.
249: 	if (colors_precomp == nullptr)
250: 	{
251: 		glm::vec3 result = computeColorFromSH(idx, D, M, (glm::vec3*)orig_points, *cam_pos, shs, clamped);
252: 		rgb[idx * C + 0] = result.x;
253: 		rgb[idx * C + 1] = result.y;
254: 		rgb[idx * C + 2] = result.z;
255: 	}
256: 
257: 	// Store some useful helper data for the next steps.
258: 	depths[idx] = p_view.z;
259: 	radii[idx] = my_radius;
260: 	points_xy_image[idx] = point_image;
261: 	// Inverse 2D covariance and opacity neatly pack into one float4
262: 	float opacity = opacities[idx];
263: 
264: 
265: 	conic_opacity[idx] = { conic.x, conic.y, conic.z, opacity * h_convolution_scaling };
266: 
267: 
268: 	tiles_touched[idx] = (rect_max.y - rect_min.y) * (rect_max.x - rect_min.x);
269: }

147: __global__ void computeCov2DCUDA(int P,
148: 	const float3* means,
149: 	const int* radii,
150: 	const float* cov3Ds,
151: 	const float h_x, float h_y,
152: 	const float tan_fovx, float tan_fovy,
153: 	const float* view_matrix,
154: 	const float* opacities,

157: 	const float* dL_dinvdepth,
158: 	float3* dL_dmeans,
159: 	float* dL_dcov,
160: 	bool antialiasing)
161: {
162: 	auto idx = cg::this_grid().thread_rank();
163: 	if (idx >= P || !(radii[idx] > 0))
164: 		return;
165: 
166: 	// Reading location of 3D covariance for this Gaussian
167: 	const float* cov3D = cov3Ds + 6 * idx;
168: 
169: 	// Fetch gradients, recompute 2D covariance and relevant 
170: 	// intermediate forward results needed in the backward.
171: 	float3 mean = means[idx];
172: 	float3 dL_dconic = { dL_dconics[4 * idx], dL_dconics[4 * idx + 1], dL_dconics[4 * idx + 3] };
173: 	float3 t = transformPoint4x3(mean, view_matrix);
174: 	
175: 	const float limx = 1.3f * tan_fovx;
176: 	const float limy = 1.3f * tan_fovy;
177: 	const float txtz = t.x / t.z;
178: 	const float tytz = t.y / t.z;
179: 	t.x = min(limx, max(-limx, txtz)) * t.z;
180: 	t.y = min(limy, max(-limy, tytz)) * t.z;
181: 	
182: 	const float x_grad_mul = txtz < -limx || txtz > limx ? 0 : 1;
183: 	const float y_grad_mul = tytz < -limy || tytz > limy ? 0 : 1;
184: 
185: 	glm::mat3 J = glm::mat3(h_x / t.z, 0.0f, -(h_x * t.x) / (t.z * t.z),
186: 		0.0f, h_y / t.z, -(h_y * t.y) / (t.z * t.z),
187: 		0, 0, 0);
188: 
189: 	glm::mat3 W = glm::mat3(
190: 		view_matrix[0], view_matrix[4], view_matrix[8],
191: 		view_matrix[1], view_matrix[5], view_matrix[9],
192: 		view_matrix[2], view_matrix[6], view_matrix[10]);
193: 
194: 	glm::mat3 Vrk = glm::mat3(
195: 		cov3D[0], cov3D[1], cov3D[2],
196: 		cov3D[1], cov3D[3], cov3D[4],
197: 		cov3D[2], cov3D[4], cov3D[5]);
198: 
199: 	glm::mat3 T = W * J;
200: 
201: 	glm::mat3 cov2D = glm::transpose(T) * glm::transpose(Vrk) * T;
202: 
203: 	// Use helper variables for 2D covariance entries. More compact.
204: 	float c_xx = cov2D[0][0];
205: 	float c_xy = cov2D[0][1];
206: 	float c_yy = cov2D[1][1];
207: 	
208: 	constexpr float h_var = 0.3f;
209: 	float d_inside_root = 0.f;
210: 	if(antialiasing)
211: 	{
212: 		const float det_cov = c_xx * c_yy - c_xy * c_xy;
213: 		c_xx += h_var;
214: 		c_yy += h_var;
215: 		const float det_cov_plus_h_cov = c_xx * c_yy - c_xy * c_xy;
216: 		const float h_convolution_scaling = sqrt(max(0.000025f, det_cov / det_cov_plus_h_cov)); // max for numerical stability
217: 		const float dL_dopacity_v = dL_dopacity[idx];
218: 		const float d_h_convolution_scaling = dL_dopacity_v * opacities[idx];
219: 		dL_dopacity[idx] = dL_dopacity_v * h_convolution_scaling;
220: 		d_inside_root = (det_cov / det_cov_plus_h_cov) <= 0.000025f ? 0.f : d_h_convolution_scaling / (2 * h_convolution_scaling);
221: 	} 
222: 	else
223: 	{
224: 		c_xx += h_var;
225: 		c_yy += h_var;
226: 	}
227: 	
228: 	float dL_dc_xx = 0;
229: 	float dL_dc_xy = 0;
230: 	float dL_dc_yy = 0;
231: 	if(antialiasing)
232: 	{
233: 		// https://www.wolframalpha.com/input?i=d+%28%28x*y+-+z%5E2%29%2F%28%28x%2Bw%29*%28y%2Bw%29+-+z%5E2%29%29+%2Fdx
234: 		// https://www.wolframalpha.com/input?i=d+%28%28x*y+-+z%5E2%29%2F%28%28x%2Bw%29*%28y%2Bw%29+-+z%5E2%29%29+%2Fdz
235: 		const float x = c_xx;
236: 		const float y = c_yy;
237: 		const float z = c_xy;
238: 		const float w = h_var;
239: 		const float denom_f = d_inside_root / sq(w * w + w * (x + y) + x * y - z * z);
240: 		const float dL_dx = w * (w * y + y * y + z * z) * denom_f;
241: 		const float dL_dy = w * (w * x + x * x + z * z) * denom_f;
242: 		const float dL_dz = -2.f * w * z * (w + x + y) * denom_f;
243: 		dL_dc_xx = dL_dx;
244: 		dL_dc_yy = dL_dy;
245: 		dL_dc_xy = dL_dz;
246: 	}
247: 	
248: 	float denom = c_xx * c_yy - c_xy * c_xy;
249: 
250: 	float denom2inv = 1.0f / ((denom * denom) + 0.0000001f);
251: 
252: 	if (denom2inv != 0)
253: 	{

255: 		// given gradients of loss w.r.t. conic matrix (inverse covariance matrix).
256: 		// e.g., dL / da = dL / d_conic_a * d_conic_a / d_a

259: 		dL_dc_yy += denom2inv * (-c_xx * c_xx * dL_dconic.z + 2 * c_xx * c_xy * dL_dconic.y + (denom - c_xx * c_yy) * dL_dconic.x);
260: 		dL_dc_xy += denom2inv * 2 * (c_xy * c_yy * dL_dconic.x - (denom + 2 * c_xy * c_xy) * dL_dconic.y + c_xx * c_xy * dL_dconic.z);
261: 		
262: 		// Gradients of loss L w.r.t. each 3D covariance matrix (Vrk) entry,
263: 		// given gradients w.r.t. 2D covariance matrix (diagonal).
264: 		// cov2D = transpose(T) * transpose(Vrk) * T;
265: 		dL_dcov[6 * idx + 0] = (T[0][0] * T[0][0] * dL_dc_xx + T[0][0] * T[1][0] * dL_dc_xy + T[1][0] * T[1][0] * dL_dc_yy);
266: 		dL_dcov[6 * idx + 3] = (T[0][1] * T[0][1] * dL_dc_xx + T[0][1] * T[1][1] * dL_dc_xy + T[1][1] * T[1][1] * dL_dc_yy);
267: 		dL_dcov[6 * idx + 5] = (T[0][2] * T[0][2] * dL_dc_xx + T[0][2] * T[1][2] * dL_dc_xy + T[1][2] * T[1][2] * dL_dc_yy);
268: 		
269: 		// Gradients of loss L w.r.t. each 3D covariance matrix (Vrk) entry,
270: 		// given gradients w.r.t. 2D covariance matrix (off-diagonal).
271: 		// Off-diagonal elements appear twice --> double the gradient.
272: 		// cov2D = transpose(T) * transpose(Vrk) * T;

274: 		dL_dcov[6 * idx + 2] = 2 * T[0][0] * T[0][2] * dL_dc_xx + (T[0][0] * T[1][2] + T[0][2] * T[1][0]) * dL_dc_xy + 2 * T[1][0] * T[1][2] * dL_dc_yy;

276: 	}
277: 	else
278: 	{
279: 		for (int i = 0; i < 6; i++)
280: 			dL_dcov[6 * idx + i] = 0;
281: 	}
